Wikidata Fetcher

Node.js script to fetch and store country data from Wikidata

This solution creates a Node.js script that fetches country data from Wikidata and stores it locally, eliminating CORS issues and improving performance.

1Create the Node.js Script

Save the following code as fetch-wikidata.js in your project directory:

const fs = require('fs'); const path = require('path'); const fetch = require('node-fetch'); // Configuration const COUNTRIES_DATA_PATH = path.join(__dirname, 'data', 'countries-basic.json'); const OUTPUT_DIR = path.join(__dirname, 'data', 'countries'); const DELAY_BETWEEN_REQUESTS = 1000; // 1 second delay between requests to be polite to the API // Ensure output directory exists if (!fs.existsSync(OUTPUT_DIR)) { fs.mkdirSync(OUTPUT_DIR, { recursive: true }); } async function fetchWikidata(wikidataId) { const url = `https://www.wikidata.org/w/api.php?action=wbgetentities&ids=${wikidataId}&format=json`; try { console.log(`Fetching data for ${wikidataId}...`); const response = await fetch(url, { headers: { 'User-Agent': 'CountryDataManager/1.0 (https://example.com)' } }); if (!response.ok) { throw new Error(`HTTP error! status: ${response.status}`); } const data = await response.json(); if (data.entities && data.entities[wikidataId]) { const entity = data.entities[wikidataId]; // Extract the information we need const result = { wikidataId: wikidataId, name: entity.labels?.en?.value || null, description: entity.descriptions?.en?.value || null, capital: extractClaimValue(entity.claims?.P36), // Capital currency: extractClaimValue(entity.claims?.P38), // Currency language: extractClaimValue(entity.claims?.P37), // Official language population: extractClaimValue(entity.claims?.P1082), // Population area: extractClaimValue(entity.claims?.P2046), // Area coordinates: extractClaimValue(entity.claims?.P625), // Coordinates flag: extractClaimValue(entity.claims?.P41) // Flag image }; return result; } else { throw new Error(`No entity found for ID: ${wikidataId}`); } } catch (error) { console.error(`Error fetching ${wikidataId}:`, error.message); return null; } } function extractClaimValue(claim) { if (!claim || !claim[0]) return null; const mainsnak = claim[0].mainsnak; if (mainsnak.datavalue) { // Handle different data types const datavalue = mainsnak.datavalue; if (datavalue.type === 'wikibase-entityid') { return datavalue.value.id; } else if (datavalue.type === 'quantity') { return datavalue.value.amount; } else if (datavalue.type === 'time') { return datavalue.value.time; } else if (datavalue.type === 'monolingualtext') { return datavalue.value.text; } else if (datavalue.type === 'string') { return datavalue.value; } else if (datavalue.type === 'globecoordinate') { return datavalue.value; } } return null; } async function fetchAllCountries() { try { // Read the country data file const rawData = fs.readFileSync(COUNTRIES_DATA_PATH); const countryData = JSON.parse(rawData); // Get all unique Wikidata IDs const wikidataIds = new Set(); const countries = countryData.countries || countryData.features || []; countries.forEach(country => { const props = country.properties || country; if (props.wikidataid) { wikidataIds.add(props.wikidataid); } }); console.log(`Found ${wikidataIds.size} countries with Wikidata IDs`); // Fetch data for each country with a delay between requests const results = []; const idsArray = Array.from(wikidataIds); for (let i = 0; i < idsArray.length; i++) { const id = idsArray[i]; const data = await fetchWikidata(id); if (data) { // Save individual country file const filename = path.join(OUTPUT_DIR, `${id}.json`); fs.writeFileSync(filename, JSON.stringify(data, null, 2)); results.push(data); } // Add delay between requests (to be polite to the API) if (i < idsArray.length - 1) { await new Promise(resolve => setTimeout(resolve, DELAY_BETWEEN_REQUESTS)); } } // Also save a combined file with all countries const combinedFilename = path.join(OUTPUT_DIR, 'all-countries.json'); fs.writeFileSync(combinedFilename, JSON.stringify(results, null, 2)); console.log(`Successfully fetched data for ${results.length} countries`); console.log(`Individual files saved in: ${OUTPUT_DIR}`); console.log(`Combined file saved as: ${combinedFilename}`); } catch (error) { console.error('Error:', error.message); } } // Run the script fetchAllCountries();

2Install Dependencies

Run the following command in your terminal to install the required dependency:

npm install node-fetch

3Run the Script

Execute the script with Node.js:

node fetch-wikidata.js

The script will:

  • Read your country data file
  • Extract all Wikidata IDs
  • Fetch detailed information for each country
  • Save individual JSON files for each country
  • Create a combined JSON file with all countries

4Modify Your Frontend

Once you have the data, update your frontend to load from the local files:

// In your CountryDataManager class, modify the fetchWikidataInfo method: async fetchWikidataInfo(wikidataId) { try { // Try to load from local file first const response = await fetch(`data/countries/${wikidataId}.json`); if (response.ok) { return await response.json(); } throw new Error('Local file not found'); } catch (error) { console.warn(`Could not load local data for ${wikidataId}:`, error); // Fallback to API if needed (though you shouldn't need it) return this.fetchFromWikidataAPI(wikidataId); } }
Note: The script includes a 1-second delay between requests to be respectful to Wikidata's servers. For a few hundred countries, this means the script will take several minutes to complete.
Scheduling: You can set up a cron job or scheduled task to run this script periodically (e.g., once a month) to keep your local data updated.
